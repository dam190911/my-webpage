3 Google Apps Script 백엔드 코드 (Code.gs)

시트 상단 메뉴 [확장 프로그램] → [Apps Script]에 아래 코드를 붙여넣고 저장한 뒤, [배포] → [새 배포] → 유형: 웹 앱 (액세스 권한: 모든 사용자)로 배포하세요.

// 반짝반짝 우리 반 포인트 상점 - Google Apps Script 백엔드 (Code.gs)

// 1. 스프레드시트를 열 때 상단 커스텀 메뉴 추가
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('🏫 우리 반 포인트 상점')
    .addItem('1. 필수 시트 3개 자동 생성 및 초기 세팅', 'setupSheets')
    .addToUi();
}

// 2. 필수 시트 3개(Students, Items, History) 자동 생성 및 스타일/샘플 세팅
function setupSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const created = [];
  const skipped = [];

  // (1) Students 시트 생성
  let stuSheet = ss.getSheetByName('Students');
  if (!stuSheet) {
    stuSheet = ss.insertSheet('Students');
    const stuHeaders = [['학생ID', '학생이름', '비밀번호(PIN)', '잔액포인트']];
    stuSheet.getRange(1, 1, 1, 4).setValues(stuHeaders);
    styleHeaderRow(stuSheet, 4, '#DBEAFE', '#1E3A8A');

    // PIN 컬럼(C열) 앞자리 0 보존을 위해 텍스트 서식(@) 지정
    stuSheet.getRange('C2:C100').setNumberFormat('@');

    const stuSamples = [
      ['S01', '김민준', '1234', 150],
      ['S02', '이서연', '0412', 230],
      ['S03', '박지훈', '0000', 80],
      ['S04', '최수아', '1004', 310],
      ['S05', '정우진', '0925', 45]
    ];
    stuSheet.getRange(2, 1, stuSamples.length, 4).setValues(stuSamples);
    stuSheet.autoResizeColumns(1, 4);
    created.push('Students');
  } else {
    skipped.push('Students');
  }

  // (2) Items 시트 생성
  let itemSheet = ss.getSheetByName('Items');
  if (!itemSheet) {
    itemSheet = ss.insertSheet('Items');
    const itemHeaders = [['상품ID', '상품명', '가격', '재고', '설명']];
    itemSheet.getRange(1, 1, 1, 5).setValues(itemHeaders);
    styleHeaderRow(itemSheet, 5, '#D1FAE5', '#065F46');

    const itemSamples = [
      ['ITEM-01', '숙제 1회 면제권', 200, 3, '알림장 기본 숙제 1회를 기분 좋게 패스할 수 있는 황금 쿠폰!'],
      ['ITEM-02', '짝꿍 자리 선택권', 150, 5, '다음 자리 바꾸기 때 내가 원하는 자리와 짝꿍을 먼저 골라요.'],
      ['ITEM-03', '급식 1등 줄서기권', 100, 10, '점심시간에 우리 반에서 가장 먼저 급식실로 출발할 수 있어요.'],
      ['ITEM-04', '선생님과 보드게임 1판', 120, 4, '쉬는 시간 또는 점심시간에 선생님과 신나는 보드게임 대결!'],
      ['ITEM-05', '달콤한 간식 뽑기권', 50, 15, '선생님의 비밀 간식 상자에서 맛있는 간식 1개를 뽑아요.']
    ];
    itemSheet.getRange(2, 1, itemSamples.length, 5).setValues(itemSamples);
    itemSheet.autoResizeColumns(1, 5);
    created.push('Items');
  } else {
    skipped.push('Items');
  }

  // (3) History 시트 생성
  let histSheet = ss.getSheetByName('History');
  if (!histSheet) {
    histSheet = ss.insertSheet('History');
    const histHeaders = [['일시', '학생이름', '구매아이템', '차감포인트', '승인상태']];
    histSheet.getRange(1, 1, 1, 5).setValues(histHeaders);
    styleHeaderRow(histSheet, 5, '#FEF3C7', '#92400E');

    const histSamples = [
      ['2026-11-22 14:20', '이서연', '달콤한 간식 뽑기권', 50, '승인완료'],
      ['2026-11-23 10:15', '김민준', '급식 1등 줄서기권', 100, '승인대기']
    ];
    histSheet.getRange(2, 1, histSamples.length, 5).setValues(histSamples);
    histSheet.autoResizeColumns(1, 5);
    created.push('History');
  } else {
    skipped.push('History');
  }

  // 기본 빈 시트('시트1' 또는 'Sheet1')가 남아있고 다른 시트가 생성되었다면 정리
  const defaultSheet = ss.getSheetByName('시트1') || ss.getSheetByName('Sheet1');
  if (defaultSheet && ss.getSheets().length > 1 && defaultSheet.getLastRow() === 0) {
    ss.deleteSheet(defaultSheet);
  }

  // UI 실행 환경일 경우 알림창 표시
  try {
    let msg = '';
    if (created.length > 0) msg += '✅ 새로 생성된 시트: ' + created.join(', ') + '\n';
    if (skipped.length > 0) msg += 'ℹ️ 이미 존재하여 보존된 시트: ' + skipped.join(', ');
    SpreadsheetApp.getUi().alert('우리 반 포인트 상점 시트 세팅 완료', msg, SpreadsheetApp.getUi().ButtonSet.OK);
  } catch (e) {
    // 웹 앱(doPost) 원격 호출 시 UI 객체가 없으므로 무시
  }

  return { created: created, skipped: skipped };
}

function styleHeaderRow(sheet, colCount, bgColor, fontColor) {
  const headerRange = sheet.getRange(1, 1, 1, colCount);
  headerRange
    .setBackground(bgColor)
    .setFontColor(fontColor)
    .setFontWeight('bold')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle');
  sheet.setRowHeight(1, 30);
  sheet.setFrozenRows(1);
}

// 3. 웹 앱 데이터 조회 (GET)
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const students = sheetToJson(ss.getSheetByName('Students'));
  const items = sheetToJson(ss.getSheetByName('Items'));
  const history = sheetToJson(ss.getSheetByName('History'));
  
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    students: students,
    items: items,
    history: history
  })).setMimeType(ContentService.MimeType.JSON);
}

// 4. 웹 앱 데이터 변경 및 원격 세팅 (POST)
function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const data = JSON.parse(e.postData.contents);
  const action = data.action;

  if (action === 'setup') {
    const res = setupSheets();
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', result: res }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === 'PURCHASE') {
    const stuSheet = ss.getSheetByName('Students');
    const itemSheet = ss.getSheetByName('Items');
    const histSheet = ss.getSheetByName('History');
    
    updateCellById(stuSheet, 0, data.studentId, 3, data.newBalance);
    updateCellById(itemSheet, 0, data.itemId, 3, data.newStock);
    histSheet.appendRow([data.timestamp, data.studentName, data.itemName, data.price, '승인대기']);
  } 
  else if (action === 'CHANGE_PIN') {
    const stuSheet = ss.getSheetByName('Students');
    updateCellById(stuSheet, 0, data.studentId, 2, "'" + data.newPin);
  }
  else if (action === 'UPDATE_POINTS') {
    const stuSheet = ss.getSheetByName('Students');
    updateCellById(stuSheet, 0, data.studentId, 3, data.newBalance);
  }
  else if (action === 'UPDATE_HISTORY_STATUS') {
    const histSheet = ss.getSheetByName('History');
    const rows = histSheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] == data.timestamp && rows[i][1] == data.studentName && rows[i][2] == data.itemName) {
        histSheet.getRange(i + 1, 5).setValue(data.status);
        break;
      }
    }
    if (data.status === '거절됨') {
      const stuSheet = ss.getSheetByName('Students');
      const itemSheet = ss.getSheetByName('Items');
      updateCellById(stuSheet, 1, data.studentName, 3, data.refundBalance);
      updateCellById(itemSheet, 1, data.itemName, 3, data.restoredStock);
    }
  }
  else if (action === 'ADD_STUDENT') {
    const stuSheet = ss.getSheetByName('Students');
    const nextRow = stuSheet.getLastRow() + 1;
    stuSheet.getRange(nextRow, 3).setNumberFormat('@');
    stuSheet.appendRow([data.id, data.name, '0000', 50]);
  }
  else if (action === 'ADD_ITEM') {
    ss.getSheetByName('Items').appendRow([data.id, data.name, data.price, data.stock, data.desc]);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function sheetToJson(sheet) {
  if (!sheet) return [];
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) return [];
  const headers = rows[0];
  return rows.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function updateCellById(sheet, idColIndex, targetId, targetColIndex, newValue) {
  if (!sheet) return;
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][idColIndex]) === String(targetId)) {
      sheet.getRange(i + 1, targetColIndex + 1).setValue(newValue);
      break;
    }
  }
}
